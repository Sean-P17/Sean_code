def preprocess(text):
    doc = nlp(text, disable=['ner', 'parser'])
    lemmas = [token.lemma_ for token in doc]
    a_lemmas = [lemma for lemma in lemmas
             if lemma.isalpha() and lemma not in stopwords]
    return ' '.join(a_lemmas)
